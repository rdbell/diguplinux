cmake_minimum_required(VERSION 3.20)
project(digup_worker LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_UI OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(MTMD_VIDEO OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_SOURCE}" llama)
add_executable(digup-worker worker.cpp)
if(GGML_VULKAN)
    target_compile_definitions(digup-worker PRIVATE DIGUP_BACKEND="vulkan")
elseif(GGML_CUDA)
    target_compile_definitions(digup-worker PRIVATE DIGUP_BACKEND="cuda")
else()
    target_compile_definitions(digup-worker PRIVATE DIGUP_BACKEND="cpu")
endif()
target_include_directories(digup-worker PRIVATE "${LLAMA_SOURCE}/vendor")
target_link_libraries(digup-worker PRIVATE llama mtmd)
